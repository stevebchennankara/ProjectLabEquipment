#include <stdio.h>

int main()
{
    char name[50][50];
    int quantity[50];
    int count = 0;
    int choice, i, sno, found;

    while (1)
    {
        printf("\n===== LAB EQUIPMENT REGISTER =====\n");
        printf("1. Add Equipment\n");
        printf("2. View All Equipment\n");
        printf("3. Search Equipment by S.No.\n");
        printf("4. Exit\n");

        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                if (count >= 50)
                {
                    printf("Equipment register is full!\n");
                }
                else
                {
                    printf("Enter equipment name: ");
                    scanf(" %49[^\n]", name[count]);

                    printf("Enter quantity: ");
                    scanf("%d", &quantity[count]);

                    count++;

                    printf("Equipment added successfully!\n");
                }
                break;

            case 2:
                if (count == 0)
                {
                    printf("No equipment records found!\n");
                }
                else
                {
                    printf("\n--- EQUIPMENT DETAILS ---\n");
                    printf("S.No.\tName\t\tQuantity\n");

                    for (i = 0; i < count; i++)
                    {
                        printf("%d\t%s\t\t%d\n",
                               i + 1, name[i], quantity[i]);
                    }
                }
                break;

            case 3:
                printf("Enter equipment S.No. to search: ");
                scanf("%d", &sno);

                found = 0;

                for (i = 0; i < count; i++)
                {
                    if (sno == i + 1)
                    {
                        printf("\nEquipment Found!\n");
                        printf("S.No.: %d\n", i + 1);
                        printf("Equipment Name: %s\n", name[i]);
                        printf("Quantity: %d\n", quantity[i]);

                        found = 1;
                        break;
                    }
                }

                if (found == 0)
                {
                    printf("Equipment not found!\n");
                }
                break;

            case 4:
                printf("Exiting Lab Equipment Register...\n");
                return 0;

            default:
                printf("Invalid choice! Please try again.\n");
        }
    }

    return 0;
}
